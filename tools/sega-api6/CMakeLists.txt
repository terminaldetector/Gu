cmake_minimum_required(VERSION 3.16)
project(genplus C)
set(CORE_DIRS core core/z80 core/m68k core/ntsc core/sound core/input_hw core/cd_hw core/cart_hw core/cart_hw/svp wasm)
foreach(dir ${CORE_DIRS})
 file(GLOB part src/main/c/${dir}/*.c)
 list(APPEND SOURCE_FILES ${part})
 include_directories(src/main/c/${dir})
endforeach()
include_directories(src/main/c/core/sound/minimp3)
add_compile_options(-O2 -std=gnu11 -fomit-frame-pointer -Wno-unused-parameter -Wno-unused-function -Wno-sign-compare)
add_definitions(-DLSB_FIRST -DUSE_32BPP_RENDERING -DHAVE_YM3438_CORE -DWASM_GENPLUS -DMAXROMSIZE=33554432)
set(CMAKE_EXE_LINKER_FLAGS "-O2 -s ALLOW_MEMORY_GROWTH=1 -s MODULARIZE=1 -s INITIAL_MEMORY=100663296 -s FILESYSTEM=0")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ../src/main/js)
add_executable(genplus ${SOURCE_FILES})
